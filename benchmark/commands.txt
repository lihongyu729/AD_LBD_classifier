# ——— 3. Group A: 元学习策略对比 (5方法 × 3种子) ————————————————
screen -S groupA
conda activate cnn
python scripts/run_batch.py --group A --seeds 42,123,456 --gpus 0 --resume

# ——— 4. Group B: 骨干网络对比 (8方法 × 3种子) ————————————————
screen -S groupB

python scripts/run_batch.py --group B --seeds 42,123,456 --gpus 0 --resume

# ——— 5. 纯5折CV（不做holdout，快速对比用） ———————————————————————
python scripts/run_batch.py --group A --seeds 42 --holdout-test 0.0 --gpus 0 --resume

# ——— 6. 消融实验（所有消融默认嵌套CV） —————————————————————————————
# Conv-only
python scripts/run_single.py --method anil --seed 42 --gpu 0 \
    --note "conv_only" \
    --set ss3m.use_dual_branch=false --set ss3m.branch_types='["conv","conv"]'

# Mamba-only
python scripts/run_single.py --method anil --seed 42 --gpu 0 \
    --note "mamba_only" \
    --set ss3m.use_dual_branch=false --set ss3m.branch_types='["mamba","mamba"]'

# 随机初始化(无预训练)
python scripts/run_single.py --method anil --seed 42 --gpu 0 \
    --note "no_pretrain" --set model.pretrained_path=""

# 等权平均融合
python scripts/run_single.py --method anil --seed 42 --gpu 0 \
    --note "avg_fusion" --set ss3m.merge_type=avg

# 方向数对比
python scripts/run_single.py --method anil --seed 42 --gpu 0 \
    --note "n_dirs_4" --set ss3m.n_dirs_train=4
python scripts/run_single.py --method anil --seed 42 --gpu 0 \
    --note "n_dirs_2" --set ss3m.n_dirs_train=2

# ——— 7. 生成对比报告 ——————————————————————————————————————————————
python scripts/generate_report.py --input ./results --output ./reports
